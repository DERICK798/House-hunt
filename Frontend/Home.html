async function requestBooking(propertyId, button) { await authReady; if (!authUser) { window.location.href = "Auth.html?mode=login&role=tenant"; return; } if (authUser.role !== "tenant") { showToast("Sign in with a tenant account to request a booking."); return; } button.disabled = true; try { const result = await bookingRequest("", { method: "POST", body: JSON.stringify({ property_id: propertyId }) }); const landlordPhone = result.booking?.landlord_phone; showToast(landlordPhone ? `Booking sent. Landlord number: ${landlordPhone}` : result.message); await loadTenantBookings(); } catch (error) { showToast(error.message || "Could not send your booking request."); button.disabled = false; } }
